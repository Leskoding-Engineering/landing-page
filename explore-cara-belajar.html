Setiap sesi berjalan seperti program yang rapi: urut, terukur, dan selalu menghasilkan output nyata.

leskoding/sesi.js
// setiap pertemuan di LesKoding
async function sesi(siswa) {
await konsep({ durasi: "15 menit", cara: "analogi + kuis" });
await praktik({ durasi: "50 menit", proyek: siswa.minat });
await feedback({ tutor: 1, maksSiswa: 5 });
return catatProgres(siswa); // XP, portofolio, rapor
}
